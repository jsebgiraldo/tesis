\begin{figure}[H]

\begin{tikzpicture}[

    halow/.style={device, fill=orange!20},
    srv/.style={device, fill=green!20},

    seglbl/.style={font=\small\bfseries, fill=white, inner sep=3pt},

    mesh/.style={thick, dashed, teal!50},

]

% ═══════════════════════════════════════════════════════
%  Segmento 1 — Nodos Thread (izquierda)
% ═══════════════════════════════════════════════════════
\node[iot] (n1) at (0, 5.5) {Nodo 1\\ESP32-C6};
\node[iot] (n2) at (0, 3.3) {Nodo 2\\ESP32-C6};
\node[iot] (n3) at (0, 1.0) {Nodo $n$\\ESP32-C6};

\draw[mesh] (n1) -- (n2);
\draw[mesh] (n2) -- (n3);
\draw[mesh] (n1) to[bend right=35] (n3);

% ═══════════════════════════════════════════════════════
%  Pasarela de borde (centro)
% ═══════════════════════════════════════════════════════
\node[gw, minimum width=3.6cm, minimum height=4.6cm] (gw1) at (5.5, 3.3) {};

\node[font=\small\bfseries, anchor=north, yshift=-4pt] at (gw1.north)
     {Pasarela de Borde};
\node[draw, rounded corners=2pt, fill=blue!8, font=\scriptsize, align=center,
      minimum width=2.9cm, minimum height=0.8cm] at (5.5, 3.8)
     {RPi4 + nRF52840\\(OTBR / otbr-agent)};
\node[draw, rounded corners=2pt, fill=orange!10, font=\scriptsize, align=center,
      minimum width=2.9cm, minimum height=0.8cm] at (5.5, 2.7)

\node[draw, rounded corners=2pt, fill=green!10, font=\scriptsize, align=center,
      minimum width=2.9cm, minimum height=0.65cm] at (5.5, 1.7)
     {TB Edge (Docker)};

% Enlace Thread → Pasarela
\draw[link, teal!70] (n1.east) -- (gw1.west |- n1.east)
     node[midway, above, annot] {LwM2M/CoAP};
\draw[link, teal!70] (n2.east) -- (gw1.west);
\draw[link, teal!70] (n3.east) -- (gw1.west |- n3.east);

%  Segmento 2 — HaLow (centro-derecha)

\draw[link] (tube.east) -- (sw.west) node[midway, above, annot] {Ethernet};
\draw[link] (sw.east)   -- (tb.west) node[midway, above, annot] {gRPC};
