\begin{verbatim}
/* Headers Zephyr */
#include <zephyr/kernel.h>
#include <zephyr/net/lwm2m.h>
#include <zephyr/net/net_mgmt.h>
#include <zephyr/net/conn_mgr_monitor.h>

/* Semáforo: espera conectividad IPv6 Thread */
static K_SEM_DEFINE(network_connected_sem, 0, 1);

/* Callback L4: Thread mesh joined → IPv6 up */
static void l4_event_handler(struct net_mgmt_event_callback *cb,
                             uint64_t event, struct net_if *iface) {
    if (event == NET_EVENT_L4_CONNECTED)
        k_sem_give(&network_connected_sem);
}

int main(void) {
    /* 1. Registrar callback conectividad L4 */
    net_mgmt_init_event_callback(&l4_cb,
        l4_event_handler, NET_EVENT_L4_CONNECTED);
    net_mgmt_add_event_callback(&l4_cb);

    /* 2. Esperar Thread join + IPv6 (credenciales en prj.conf) */
    k_sem_take(&network_connected_sem, K_FOREVER);

    /* 3. Configurar objetos LwM2M (Security, Server, Device, IPSO) */
    lwm2m_set_string(&LWM2M_OBJ(0, 0, 0),     /* Security URI */
        "coap://[" CONFIG_NET_CONFIG_PEER_IPV6_ADDR "]:5683");
    lwm2m_set_string(&LWM2M_OBJ(3, 0, 0),      /* Device Manufacturer */
        "Tesis-AMI");

    /* 4. Registrar cliente con servidor Leshan */
    lwm2m_rd_client_start(&client_ctx, "ami-esp32c6-001",
                          0, rd_client_event, NULL);

    /* 5. Ciclo principal: actualizar sensores cada 30 s */
    while (1) {
        lwm2m_set_f64(&LWM2M_OBJ(3316, 0, 5700), voltage_v);
        lwm2m_set_f64(&LWM2M_OBJ(3317, 0, 5700), current_a);
        k_sleep(K_SECONDS(30));
    }
    return 0;
}

\section{Arquitectura del Nodo}

\subsection{Hardware}

\begin{itemize}
    \item \textbf{MCU}: ESP32-C6 (RISC-V, 160 MHz, 512 KB SRAM)
    \item \textbf{Radio}: IEEE 802.15.4 (Thread 1.3) integrado
    \item \textbf{Interfaz medición}: RS-485 UART (DLMS/COSEM IEC 62056-21)
    \item \textbf{Alimentación}: Batería Li-Ion 18650 3.7V + regulador 3.3V
    \item \textbf{Modos de bajo consumo}: Deep sleep (<20 $\mu$A), light sleep (~800 $\mu$A)
\end{itemize}

\subsection{Stack de Software}

\begin{itemize}
    \item \textbf{Framework}: nRF Connect SDK 2.6.0 (Zephyr RTOS 3.5)
    \item \textbf{Pila Thread}: OpenThread (Joiner commissioning)
    \item \textbf{Pila LwM2M}: Zephyr LwM2M engine nativo (cliente LwM2M 1.1)
    \item \textbf{Objetos LwM2M}: IPSO Smart Objects (3316 Voltage, 3317 Current, 3305 Power Measurement, 3300 Generic Sensor), Device (3), Connectivity (4), Location (6)
    \item \textbf{Recursos IPSO}: 5700 (Sensor Value), 5701 (Sensor Units), 5601 (Min Value), 5602 (Max Value) por instancia/fase
    \item \textbf{Transporte}: CoAP sobre UDP/IPv6 (Thread)
\end{itemize}

#include <stdio.h>
#include <zephyr/kernel.h>
#include <zephyr/logging/log.h>
#include <zephyr/settings/settings.h>
#include <zephyr/drivers/gpio.h>

// Módulos locales
#include "thread_prov.h"
#include "led_status.h"

LOG_MODULE_REGISTER(lwm2m_main, LOG_LEVEL_INF);

void lwm2m_client_start(void);

static const char *TAG = "lwm2m_main";

// GPIO para botón de factory reset (ESP32-C6: GPIO9 típico)
#define CONFIG_BOARD_BOOT_BUTTON_GPIO 9
#define CONFIG_FACTORY_RESET_HOLD_MS 5000

static inline bool is_deep_sleep_wake_capable_gpio(gpio_num_t gpio)
{
    // En ESP32-C6, GPIO0-GPIO7 son LP GPIOs (wake from deep sleep)
    return (gpio >= GPIO_NUM_0 && gpio <= GPIO_NUM_7);
}

static void factory_reset_task(void* arg)
{
    const gpio_num_t btn = (gpio_num_t)CONFIG_BOARD_BOOT_BUTTON_GPIO;
    const TickType_t hold_ticks = pdMS_TO_TICKS(CONFIG_FACTORY_RESET_HOLD_MS);
    
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << btn),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    gpio_config(&io_conf);
    
    while (1) {
        if (gpio_get_level(btn) == 0) {  // Botón presionado (activo bajo)
            TickType_t press_start = xTaskGetTickCount();
            
            while (gpio_get_level(btn) == 0) {
                TickType_t elapsed = xTaskGetTickCount() - press_start;
                if (elapsed >= hold_ticks) {
                    ESP_LOGW(TAG, "Factory reset triggered! Erasing NVS...");
                    
                    // Parpadeo LED rápido para indicar reset
                    led_status_factory_reset();
                    
                    // Borrar partición NVS
                    nvs_flash_erase();
                    nvs_flash_init();
                    
                    ESP_LOGW(TAG, "Factory reset complete. Rebooting...");
                    vTaskDelay(pdMS_TO_TICKS(1000));
                    esp_restart();
                }
                vTaskDelay(pdMS_TO_TICKS(100));
            }
        }
        vTaskDelay(pdMS_TO_TICKS(200));
    }
}

void app_main(void)
{
    ESP_LOGI(TAG, "=== LwM2M Temperature/Humidity Node ===");
    ESP_LOGI(TAG, "ESP-IDF version: %s", esp_get_idf_version());
    
    // Inicializar NVS (almacenamiento persistente)
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || 
        ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);
    
    // Inicializar LED de estado
    led_status_init();
    led_status_set(LED_STATUS_BOOTING);
    
    // Iniciar tarea de factory reset en background
    xTaskCreate(factory_reset_task, "factory_rst", 2048, NULL, 
                tskIDLE_PRIORITY + 1, NULL);
    
#if CONFIG_LWM2M_NETWORK_USE_THREAD
    ESP_LOGI(TAG, "Starting Thread Provisioning...");
    thread_provisioning_init();
    
    ESP_LOGI(TAG, "Waiting for Thread network attachment...");
    thread_provisioning_wait_connected();
    
    ESP_LOGI(TAG, "Thread connected! Starting LwM2M client...");
    led_status_set(LED_STATUS_CONNECTED);
    lwm2m_client_start();
    
#elif CONFIG_LWM2M_NETWORK_USE_WIFI
    ESP_LOGI(TAG, "Starting WiFi Provisioning...");
    wifi_provisioning_init();
    
    ESP_LOGI(TAG, "Waiting for WiFi connection...");
    wifi_provisioning_wait_connected();
    
    ESP_LOGI(TAG, "WiFi connected! Starting LwM2M client...");
    led_status_set(LED_STATUS_CONNECTED);
    lwm2m_client_start();
    
#else
    ESP_LOGE(TAG, "No network backend enabled. "
                  "Enable Thread or WiFi in menuconfig.");
    led_status_set(LED_STATUS_ERROR);
#endif
}

\section{Cliente LwM2M}

\subsection{lwm2m\_client.c (fragmento principal)}

Cliente LwM2M Zephyr con registro de objetos IPSO y manejo de eventos:

\begin{verbatim}
#include <zephyr/kernel.h>
#include <zephyr/logging/log.h>
#include <zephyr/net/net_if.h>
#include <string.h>
#include <stdlib.h>

// Zephyr LwM2M engine nativo
#include <zephyr/net/lwm2m.h>
#include <zephyr/net/lwm2m_path.h>

LOG_MODULE_REGISTER(lwm2m_client, LOG_LEVEL_INF);

static const char *TAG = "lwm2m_client";

// Endpoint name (único por dispositivo, basado en MAC)
static char g_endpoint_name[32] = {0};

static struct lwm2m_ctx client_ctx;

static void resolve_endpoint_name(void)
{
    if (strlen(g_endpoint_name) > 0) {
        return;  // Ya resuelto
    }

#ifdef CONFIG_LWM2M_ENDPOINT_NAME
    strncpy(g_endpoint_name, CONFIG_LWM2M_ENDPOINT_NAME,
            sizeof(g_endpoint_name) - 1);
#else
    // Generar desde MAC address
    uint8_t mac[6];
    esp_efuse_mac_get_default(mac);
    snprintf(g_endpoint_name, sizeof(g_endpoint_name),
             "esp32c6_%02x%02x%02x", mac[3], mac[4], mac[5]);
#endif
}

// Crear instancias IPSO Smart Objects para medición trifásica
static int create_ipso_objects(void)
{
    int ret;

    // --- IPSO Voltage (3316): una instancia por fase ---
    ret = lwm2m_engine_create_obj_inst("3316/0");  // Voltage L1
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3316/0: %d", ret); return ret; }
    lwm2m_engine_set_string("3316/0/5701", "V");

    ret = lwm2m_engine_create_obj_inst("3316/1");  // Voltage L2
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3316/1: %d", ret); return ret; }
    lwm2m_engine_set_string("3316/1/5701", "V");

    ret = lwm2m_engine_create_obj_inst("3316/2");  // Voltage L3
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3316/2: %d", ret); return ret; }
    lwm2m_engine_set_string("3316/2/5701", "V");

    // --- IPSO Current (3317): una instancia por fase ---
    ret = lwm2m_engine_create_obj_inst("3317/0");  // Current L1
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3317/0: %d", ret); return ret; }
    lwm2m_engine_set_string("3317/0/5701", "A");

    ret = lwm2m_engine_create_obj_inst("3317/1");  // Current L2
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3317/1: %d", ret); return ret; }
    lwm2m_engine_set_string("3317/1/5701", "A");

    ret = lwm2m_engine_create_obj_inst("3317/2");  // Current L3
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3317/2: %d", ret); return ret; }
    lwm2m_engine_set_string("3317/2/5701", "A");

    // --- IPSO Power Measurement (3305): potencia activa/reactiva ---
    ret = lwm2m_engine_create_obj_inst("3305/0");  // Active Power
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3305/0: %d", ret); return ret; }
    lwm2m_engine_set_string("3305/0/5701", "kW");

    ret = lwm2m_engine_create_obj_inst("3305/1");  // Reactive Power
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3305/1: %d", ret); return ret; }
    lwm2m_engine_set_string("3305/1/5701", "kvar");

    // --- IPSO Generic Sensor (3300): energía acumulada, frecuencia, PF ---
    ret = lwm2m_engine_create_obj_inst("3300/0");  // Active Energy
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3300/0: %d", ret); return ret; }
    lwm2m_engine_set_string("3300/0/5701", "kWh");

    ret = lwm2m_engine_create_obj_inst("3300/1");  // Frequency
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3300/1: %d", ret); return ret; }
    lwm2m_engine_set_string("3300/1/5701", "Hz");

    ret = lwm2m_engine_create_obj_inst("3300/2");  // Power Factor
    if (ret < 0) { ESP_LOGE(TAG, "Failed 3300/2: %d", ret); return ret; }
    lwm2m_engine_set_string("3300/2/5701", "");

    ESP_LOGI(TAG, "IPSO objects created: 3316(V)x3, 3317(A)x3, "
                  "3305(P)x2, 3300(E/Hz/PF)x3");
    return 0;
}

// Configurar Security (Object 0) y Server (Object 1)
static int setup_security_and_server(void)
{
    // Security Object (0): NoSec sobre red Thread confiable
    lwm2m_engine_set_string("0/0/0",
        CONFIG_LWM2M_SERVER_URI);           // coap://[fd00::1]:5683
    lwm2m_engine_set_u8("0/0/2", 3);       // Security Mode: NoSec

    // Server Object (1): lifetime y binding
    lwm2m_engine_set_u32("1/0/1", 300);    // Lifetime 5 min
    lwm2m_engine_set_string("1/0/7", "U"); // Binding: UDP

    ESP_LOGI(TAG, "Security/Server configured: URI=%s Lifetime=300s",
             CONFIG_LWM2M_SERVER_URI);
    return 0;
}

static void rd_client_event_cb(struct lwm2m_ctx *ctx,
                               enum lwm2m_rd_client_event event)
{
    switch (event) {
    case LWM2M_RD_CLIENT_EVENT_REGISTRATION_COMPLETE:
        ESP_LOGI(TAG, "LwM2M Registration complete");
        break;
    case LWM2M_RD_CLIENT_EVENT_DISCONNECT:
        ESP_LOGW(TAG, "LwM2M Disconnected");
        break;
    case LWM2M_RD_CLIENT_EVENT_REG_UPDATE_COMPLETE:
        ESP_LOGD(TAG, "Registration update complete");
        break;
    default:
        ESP_LOGD(TAG, "RD client event: %d", event);
        break;
    }
}

static void lwm2m_client_task(void *arg)
{
    resolve_endpoint_name();
    ESP_LOGI(TAG, "LwM2M Endpoint: %s", g_endpoint_name);

    // Configurar objetos Security/Server
    if (setup_security_and_server()) {
        ESP_LOGE(TAG, "Failed to configure Security/Server");
        vTaskDelete(NULL);
    }

    // Crear instancias IPSO Smart Objects
    if (create_ipso_objects()) {
        ESP_LOGE(TAG, "Failed to create IPSO objects");
        vTaskDelete(NULL);
    }

    // Iniciar cliente RD (Registration)
    memset(&client_ctx, 0, sizeof(client_ctx));
    lwm2m_rd_client_start(&client_ctx, g_endpoint_name,
                          0,  // flags
                          rd_client_event_cb);

    ESP_LOGI(TAG, "Zephyr LwM2M RD client started");

    // Loop principal: actualizar valores de sensores
    while (1) {
        vTaskDelay(pdMS_TO_TICKS(1000));  // Muestreo cada 1s

        // Actualizar lecturas DLMS/COSEM (ver ipso_power_update.c)
        ipso_power_update();
    }
}

void lwm2m_client_start(void)
{
    xTaskCreate(lwm2m_client_task, "lwm2m",
                CONFIG_LWM2M_TASK_STACK_SIZE,  // 8192
                NULL, tskIDLE_PRIORITY + 2, NULL);
}
\end{verbatim}

\section{Objetos IPSO Smart Objects}

\subsection{ipso\_power\_update.c}

Implementación de actualización de objetos IPSO Smart Objects con lecturas DLMS/COSEM~\cite{dlmsLwM2MIntegration2024}:

\begin{verbatim}
#include <zephyr/kernel.h>
#include <zephyr/net/lwm2m.h>
#include <zephyr/logging/log.h>
#include <zephyr/drivers/uart.h>  // RS-485 DLMS/COSEM
#include <math.h>
#include <stdbool.h>

LOG_MODULE_REGISTER(ipso_power, LOG_LEVEL_INF);

// IPSO Object IDs
#define OID_VOLTAGE    3316   // IPSO Voltage
#define OID_CURRENT    3317   // IPSO Current
#define OID_POWER      3305   // IPSO Power Measurement
#define OID_GENERIC    3300   // IPSO Generic Sensor

// IPSO Resource IDs (comunes a todos los objetos IPSO)
#define RID_SENSOR_VALUE  5700   // Sensor Value (float)
#define RID_SENSOR_UNITS  5701   // Sensor Units (string)
#define RID_MIN_VALUE     5601   // Min Measured Value
#define RID_MAX_VALUE     5602   // Max Measured Value

#define METER_SAMPLE_INTERVAL_MS 1000
#define UART_DLMS_NUM UART_NUM_1  // RS-485 en GPIO 16/17

static const char *TAG = "ipso_power";

// Estado interno (lecturas DLMS/COSEM)
static float g_voltage[3] = {230.0f, 230.0f, 230.0f};  // V por fase
static float g_current[3] = {0.0f, 0.0f, 0.0f};        // A por fase
static float g_active_power = 0.0f;     // kW
static float g_reactive_power = 0.0f;   // kvar
static float g_power_factor = 1.0f;     // cos(phi)
static float g_active_energy = 0.0f;    // kWh acumulada
static float g_frequency = 60.0f;       // Hz
static TickType_t g_last_sample_tick = 0;

// Leer medidor DLMS/COSEM vía RS-485 (OBIS codes IEC 62056-21)
static int read_power_meter_dlms(void)
{
    TickType_t ticks = xTaskGetTickCount();
    float phase = (float)(ticks % 60000) / 10000.0f;

    // Voltajes por fase: 230V ±5V
    for (int i = 0; i < 3; i++) {
        g_voltage[i] = 230.0f +
            ((float)(esp_random() % 100) / 10.0f) - 5.0f;
    }

    // Potencia activa: 2-5 kW con variación
    float base_power = 3.0f;
    g_active_power = base_power + 1.5f * sinf(phase) +
                     ((float)(esp_random() % 100) / 1000.0f);

    g_power_factor = 0.92f +
                     ((float)(esp_random() % 80) / 1000.0f);

    // Corrientes por fase: I = P / (3 * V * PF)
    for (int i = 0; i < 3; i++) {
        g_current[i] = (g_active_power * 1000.0f) /
                        (3.0f * g_voltage[i] * g_power_factor);
    }

    // Potencia reactiva: Q = P * tan(acos(PF))
    float angle = acosf(g_power_factor);
    g_reactive_power = g_active_power * tanf(angle);

    // Energía acumulada (integración Riemann 1s)
    g_active_energy += g_active_power / 3600.0f;  // kWh

    g_frequency = 60.0f +
        ((float)(esp_random() % 100) / 1000.0f) - 0.05f;

    return 0;
}

// Actualizar todos los objetos IPSO con nuevas lecturas
void ipso_power_update(void)
{
    TickType_t now = xTaskGetTickCount();
    if (g_last_sample_tick != 0 &&
        (now - g_last_sample_tick) < pdMS_TO_TICKS(METER_SAMPLE_INTERVAL_MS)) {
        return;  // No ha pasado el intervalo de muestreo
    }

    read_power_meter_dlms();
    g_last_sample_tick = now;

    // --- Voltajes IPSO 3316 (instancias 0, 1, 2 = L1, L2, L3) ---
    for (int i = 0; i < 3; i++) {
        char path[24];
        snprintf(path, sizeof(path), "3316/%d/5700", i);
        lwm2m_engine_set_float(path, &g_voltage[i]);
    }

    // --- Corrientes IPSO 3317 (instancias 0, 1, 2 = L1, L2, L3) ---
    for (int i = 0; i < 3; i++) {
        char path[24];
        snprintf(path, sizeof(path), "3317/%d/5700", i);
        lwm2m_engine_set_float(path, &g_current[i]);
    }

    // --- Potencias IPSO 3305 ---
    lwm2m_engine_set_float("3305/0/5700", &g_active_power);
    lwm2m_engine_set_float("3305/1/5700", &g_reactive_power);

    // --- Sensores genéricos IPSO 3300 ---
    lwm2m_engine_set_float("3300/0/5700", &g_active_energy);
    lwm2m_engine_set_float("3300/1/5700", &g_frequency);
    lwm2m_engine_set_float("3300/2/5700", &g_power_factor);

    ESP_LOGD(TAG, "IPSO update: V=%.1f/%.1f/%.1f  I=%.2f/%.2f/%.2f  "
                  "P=%.3fkW  E=%.3fkWh",
             g_voltage[0], g_voltage[1], g_voltage[2],
             g_current[0], g_current[1], g_current[2],
             g_active_power, g_active_energy);
}

             DEVICE_MANUFACTURER, DEVICE_MODEL);
}

// Callback para Execute en recurso Reboot (3/0/4)
static int device_reboot_cb(uint16_t obj_inst_id,
                            uint8_t *args, uint16_t args_len)
{
    ESP_LOGW(TAG, "Reboot requested via LwM2M");
    esp_restart();
    return 0;  // No retorna
}

void device_object_register_callbacks(void)
{
    lwm2m_engine_register_exec_callback("3/0/4",
                                        device_reboot_cb);
}

void device_object_update(void)
{
    static TickType_t last_update = 0;
    TickType_t now = xTaskGetTickCount();

    if ((now - last_update) >= pdMS_TO_TICKS(10000)) {
        last_update = now;

        // Actualizar memoria libre
        int32_t free_mem = (int32_t)esp_get_free_heap_size();
        lwm2m_engine_set_s32("3/0/10", free_mem);

        // Actualizar nivel de batería simulado (70-100%)
        static int32_t battery = 100;
        battery -= 1;
        if (battery < 70) battery = 100;
        lwm2m_engine_set_s32("3/0/9", battery);
    }
}

\section{Conectividad Thread}

\subsection{thread\_prov.c (fragmento)}

Provisioning de red Thread con OpenThread Joiner:

\begin{verbatim}
#include "thread_prov.h"
#include <string.h>
#include <esp_log.h>
#include <esp_openthread.h>
#include <esp_openthread_lock.h>
#include <openthread/thread.h>
#include <openthread/joiner.h>

static const char *TAG = "thread_prov";

static void ot_joiner_callback(otError error, void *context)
{
    if (error == OT_ERROR_NONE) {
        ESP_LOGI(TAG, "Joiner success! Attached to Thread network");
        
        esp_openthread_lock_acquire(portMAX_DELAY);
        otThreadSetEnabled(esp_openthread_get_instance(), true);
        esp_openthread_lock_release();
    } else {
        ESP_LOGE(TAG, "Joiner failed: %d", error);
    }
}

void thread_provisioning_init(void)
{
    ESP_LOGI(TAG, "Initializing OpenThread...");
    
    // Configuración Thread por defecto
    esp_openthread_platform_config_t config = {
        .radio_config = ESP_OPENTHREAD_DEFAULT_RADIO_CONFIG(),
        .host_config = ESP_OPENTHREAD_DEFAULT_HOST_CONFIG(),
        .port_config = ESP_OPENTHREAD_DEFAULT_PORT_CONFIG(),
    };
    
    ESP_ERROR_CHECK(esp_openthread_init(&config));
    
    otInstance *instance = esp_openthread_get_instance();
    
    // Iniciar Joiner con PSKd (pre-shared key for device)
    esp_openthread_lock_acquire(portMAX_DELAY);
    
    const char *pskd = CONFIG_THREAD_JOINER_PSKD;  // "J01NME"
    otError error = otJoinerStart(instance, pskd, NULL, PACKAGE_NAME,
                                   NULL, NULL, NULL, 
                                   ot_joiner_callback, NULL);
    
    esp_openthread_lock_release();
    
    if (error != OT_ERROR_NONE) {
        ESP_LOGE(TAG, "Failed to start Joiner: %d", error);
    } else {
        ESP_LOGI(TAG, "Joiner started with PSKd");
    }
}

void thread_provisioning_wait_connected(void)
{
    ESP_LOGI(TAG, "Waiting for Thread attachment...");
    
    while (1) {
        esp_openthread_lock_acquire(portMAX_DELAY);
        otInstance *instance = esp_openthread_get_instance();
        otDeviceRole role = otThreadGetDeviceRole(instance);
        esp_openthread_lock_release();
        
        if (role >= OT_DEVICE_ROLE_CHILD) {
            ESP_LOGI(TAG, "Thread attached! Role: %d", role);
            break;
        }
        
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
\end{verbatim}

    \item \textit{Attributes} mostrará: manufacturer, model, fw\_version, serial\_number
    \item Configurar \textit{Observe} en recursos 3305/0/5700 (Active Power) y 3300/0/5700 (Active Energy) para notificaciones automáticas con \texttt{pmin=60\&pmax=900\&st=0.05}
\end{enumerate}
