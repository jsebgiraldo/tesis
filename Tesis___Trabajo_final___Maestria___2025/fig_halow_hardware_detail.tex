% Diagrama detallado de conexion hardware Alfa Networks Tube AHM con Raspberry Pi 4

\begin{tikzpicture}[

    antenna/.style={isosceles triangle, draw=red!60, fill=red!20, thick, minimum width=0.5cm, minimum height=1cm, rotate=90},
    connector/.style={rectangle, draw=black, fill=gray!30, thick, minimum width=1.5cm, minimum height=0.4cm, font=\tiny},
    arrow/.style={->, thick, >=stealth}
]

% ============ ALFA NETWORKS TUBE AHM (AP MESH) ============

\node[device, fill=green!5] (tube) at (0,4) {};
\node[above=-0.1cm of tube.north, font=\bfseries] {Alfa Networks Tube AHM};

% Componentes internos Tube AHM
\node[module, fill=yellow!20] at (0,5) (mt7628) {
    \textbf{MT7628}\\
    \tiny MIPS 580 MHz\\OpenWRT 23.05

\node[antenna, fill=red!40] at (1.5,4.5) (ant_tube) {};
\node[right=0.1cm of ant_tube, font=\tiny] {5 dBi omni};

% Puerto Ethernet Tube AHM
\node[connector] at (0,2.5) (eth_tube) {ETH0\\PoE 24V};

% Etiqueta role
\node[below=0.2cm of eth_tube, font=\footnotesize, text=green!50!black] {
    \textbf{Role: Mesh Point}\\(modo mesh 802.11s HWMP)
};

% ============ RASPBERRY PI 4 (GATEWAY EDGE INTEGRADO) ============

\node[device, fill=orange!5] (rpi) at (0,-2) {};
\node[above=-0.1cm of rpi.north, font=\bfseries] {Raspberry Pi 4 (Gateway Edge)};

    \textbf{nRF52840}\\

};

\node[connector] at (0,-3.2) (eth_rpi) {ETH0\\GbE (primary)

\draw[arrow, black, thick] (eth_rpi) -- (eth_tube) 
    node[midway, right, font=\small, text width=3cm, align=left] {
        \textbf{Ethernet Local}\\

        RPi4 inyecta tráfico\\
        agregado Thread al\\
        backbone HaLow mesh
    };

% ============ NOTAS ARQUITECTURA ============

\node[below=0.5cm of rpi, font=\footnotesize, text width=12cm, align=left] {
    $\bullet$ \textbf{Arquitectura Distributed Mesh Gateways (Peer-to-Peer):}\\[0.2cm]
    $\bullet$ \textbf{Alfa Networks Tube AHM:} Router comercial standalone MT7628 + MM6108 radio HaLow 802.11ah. Opera en modo Mesh Point IEEE 802.11s peer con protocolo HWMP, sin jerarquía maestro-esclavo.\\[0.1cm]
    $\bullet$ \textbf{Raspberry Pi 4:} Gateway edge con Docker stack (ThingsBoard Edge, OTBR, PostgreSQL). Integrado localmente por Ethernet GbE al puerto LAN del Tube AHM co-ubicado.\\[0.1cm]
    $\bullet$ \textbf{Conexión Ethernet local:} RPi4 conectado directamente por Ethernet al Tube AHM. Tráfico Thread agregado inyectado al backbone HaLow mesh compartido vía routing HWMP cooperativo.\\[0.1cm]
    $\bullet$ \textbf{Ventaja peer-to-peer:} Todos los Gateways Edge son Mesh Points peers equipolentes. Múltiples rutas redundantes entre gateways hacia Mesh Gate. Fallos de cualquier Mesh Point no interrumpen servicio de otros: HWMP re-enruta tráfico automáticamente por paths alternativos disponibles en topología full-mesh.\\[0.1cm]
    $\bullet$ \textbf{Thread Border Router:} Cada RPi4 ejecuta OTBR mediante nRF52840 USB Dongle, creando red Thread local para 15 nodos IoT ESP32-C6.
};

\end{tikzpicture}
\caption{Detalle hardware de integración Gateway Edge en Mesh Point peer: Alfa Networks Tube AHM (Mesh Point) y Raspberry Pi 4 (Gateway Edge) co-ubicados. El Tube AHM integra procesador MT7628 con radio MM6108 actuando como Mesh Point peer activo IEEE 802.11s sin jerarquía maestro-esclavo. El RPi4 se conecta localmente por Ethernet GbE al puerto LAN del Tube AHM, inyectando tráfico Thread agregado al backbone HaLow mesh compartido. Arquitectura distributed mesh gateways peer-to-peer: todos los gateways edge son Mesh Points equipolentes que participan cooperativamente en routing HWMP, habilitando múltiples paths redundantes y re-enrutamiento automático ante fallos, mejorando disponibilidad vs arquitectura cliente STA tradicional.}
\label{fig:halow-hardware-detail}
\end{figure}
